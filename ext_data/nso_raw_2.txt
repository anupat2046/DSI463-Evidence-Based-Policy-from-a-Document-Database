<script type="text/javascript">
  $(document).ready(function() {
    // ป้องกันการพิมพ์ภาษาไทย หรืออักขระที่ไม่ใช่ภาษาอังกฤษ/ตัวเลข/สัญลักษณ์พื้นฐาน
    $('#user_id, #password').on('input', function() {
      this.value = this.value.replace(/[^\x20-\x7E]/g, '');
    });

    // ระบบกดดูลูกตารหัสผ่าน
    $('.toggle-password').click(function() {
      var targetId = $(this).data('target');
      var input = $('#' + targetId);
      if (input.attr('type') === 'password') {
        input.attr('type', 'text');
        $(this).removeClass('fa-eye').addClass('fa-eye-slash');
      } else {
        input.attr('type', 'password');
        $(this).removeClass('fa-eye-slash').addClass('fa-eye');
      }
    });

    $('.login100-form').submit(function(e) {
      e.preventDefault();
      var user_id = $("#user_id").val();
      var password = $("#password").val();
      var project_id = $("#project_id").val();
      var btn = $("#btn_login");

      if (!user_id || !password) {
        Swal.fire({
          icon: 'warning',
          title: 'แจ้งเตือน',
          text: 'กรุณากรอกรหัสผู้ใช้และรหัสผ่าน',
          confirmButtonText: 'ตกลง',
          confirmButtonColor: '#4776E6'
        });
        return;
      }

      btn.prop('disabled', true).html('<i class="fa fa-spinner fa-spin"></i> กำลังตรวจสอบ...');

      $.ajax({
        type: "POST",
        url: "check_login.php",
        dataType: "json",
        data: {
          user_id: user_id,
          password: password,
          project_id: project_id,
          ajax: 1
        },
        success: function(response) {
          if (response.status === 'success') {
            Swal.fire({
              icon: 'success',
              title: 'เข้าสู่ระบบสำเร็จ',
              text: 'กำลังพาท่านเข้าสู่ระบบ...',
              showConfirmButton: false,
              timer: 1500
            }).then(() => {
              window.location.href = response.url;
            });
          } else {
            Swal.fire({
              icon: 'error',
              title: 'เข้าสู่ระบบไม่สำเร็จ',
              text: response.message,
              confirmButtonText: 'ตกลง',
              confirmButtonColor: '#e73c7e'
            });
            btn.prop('disabled', false).html('เข้าสู่ระบบ');
          }
        },
        error: function() {
            Swal.fire({
              icon: 'error',
              title: 'เกิดข้อผิดพลาด',
              text: 'ไม่สามารถติดต่อเซิร์ฟเวอร์ได้',
              confirmButtonText: 'ตกลง',
              confirmButtonColor: '#e73c7e'
            });
            btn.prop('disabled', false).html('เข้าสู่ระบบ');
        }
      });
    });
  });
</script>
